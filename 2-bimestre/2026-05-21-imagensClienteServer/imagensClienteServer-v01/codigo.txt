        // Nova função que lê o arquivo local e exibe na tela antes/depois do upload
        function mostrarPreview(event) {
            const input = event.target;
            const preview = document.getElementById('previewImagem');
            
            if (input.files && input.files[0]) {
                const reader = new FileReader();
                
                // Quando o navegador terminar de processar a imagem localmente, define o src
                reader.onload = function(e) {
                    preview.src = e.target.result;
                    preview.style.display = 'block'; // Torna a imagem visível
                }
                
                reader.readAsDataURL(input.files[0]);
            } else {
                preview.src = '';
                preview.style.display = 'none';
            }
        }

        // --- FUNÇÃO PARA ENVIAR IMAGEM ---
        async function enviarImagem() {
            const enderecoServidor = document.getElementById('enderecoServidorInput').value;
            const porta = document.getElementById('inputPorta').value;
            const inputImagem = document.getElementById('inputImagem');
            const respostaDiv = document.getElementById("respostaImagem");

            if (inputImagem.files.length === 0) {
                respostaDiv.className = "resposta aviso"; // Classe amarela
                respostaDiv.innerHTML = "Por favor, selecione uma imagem primeiro!";
                return;
            }

            const arquivoImagem = inputImagem.files[0];
            const formData = new FormData();
            formData.append('foto', arquivoImagem);

            try {
                respostaDiv.className = "resposta";
                respostaDiv.innerHTML = "Enviando arquivo...";

                const response = await fetch(`http://${enderecoServidor}:${porta}/enviar-imagem`, {
                    method: 'POST',
                    body: formData
                });

                if (!response.ok) {
                    throw new Error('Falha na resposta do servidor');
                }

                const data = await response.json();

                // Define a classe de sucesso (Verde) e atualiza o texto permanentemente
                respostaDiv.className = "resposta sucesso";
                respostaDiv.innerHTML = `<b>${data.mensagem}</b><br>Salvo no servidor como: <i>${data.nomeArquivo}</i>`;

            } catch (error) {
                console.error('Erro:', error);
                // Define a classe de erro (Vermelho)
                respostaDiv.className = "resposta erro";
                respostaDiv.innerHTML = "Erro de comunicação: Não foi possível salvar a imagem no servidor.";
            }
        }

        // --- FUNÇÃO DE PITÁGORAS ---
        async function enviarMensagem() {            
            const enderecoServidor = document.getElementById('enderecoServidorInput').value;
            const porta = document.getElementById('inputPorta').value;
            const respostaDiv = document.getElementById("respostaDoServidor");
            
            const c1 = parseFloat(document.getElementById('inputCateto1').value);
            const c2 = parseFloat(document.getElementById('inputCateto2').value);

            if (isNaN(c1) || isNaN(c2)) {
                respostaDiv.className = "resposta aviso"; // Amarelo
                respostaDiv.innerHTML = "Por favor, insira valores numéricos válidos nos dois catetos.";
                return;
            }

            const meuPacoteDeDados = { x: c1, y: c2 };
            
            try {
                respostaDiv.className = "resposta";
                respostaDiv.innerHTML = "Processando cálculo...";

                const response = await fetch(`http://${enderecoServidor}:${porta}/enviar-mensagem`, {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                    },
                    body: JSON.stringify(meuPacoteDeDados),
                });
                
                const data = await response.text();
                const { hipotenusa } = JSON.parse(data); 
                
                respostaDiv.className = "resposta sucesso"; // Verde
                respostaDiv.innerHTML = "Cálculo concluído! A hipotenusa é: <b>" + hipotenusa + "</b>";
            } catch (error) {
                console.error('Erro:', error);
                respostaDiv.className = "resposta erro"; // Vermelho
                respostaDiv.innerHTML = "Erro ao se comunicar com o servidor para calcular a hipotenusa.";
            }
        }

const express = require('express');
const os = require('os');
const path = require('path');
const fs = require('fs');
const multer = require('multer');

const app = express();
const port = 3000;

// Configuração do diretório onde as imagens serão salvas
const pastaImagens = './imagens/';

// Garante que a pasta existe, se não existir, cria ela
if (!fs.existsSync(pastaImagens)){
    fs.mkdirSync(pastaImagens, { recursive: true });
}

// Configuração de armazenamento do Multer
const armazenamento = multer.diskStorage({
    destination: (req, file, cb) => {
        cb(null, pastaImagens); // Define a pasta de destino
    },
    filename: (req, file, cb) => {
        // Define o nome do arquivo: timestamp atual + nome original do arquivo para evitar duplicatas
        const nomeUnico = Date.now() + '-' + file.originalname;
        cb(null, nomeUnico);
    }
});

// Inicializa o middleware do multer com a configuração de armazenamento
const upload = multer({ storage: armazenamento });

// Middlewares padrão
app.use(express.json());

// Middleware CORS adaptado para suportar o FormData e métodos necessários
app.use((req, res, next) => {
    res.header('Access-Control-Allow-Origin', '*');
    res.header('Access-Control-Allow-Methods', 'POST, GET, OPTIONS');
    res.header('Access-Control-Allow-Headers', 'Content-Type');
    next();
});

// --- ROTA ANTERIOR (Mantida para exemplo) ---
app.post('/enviar-mensagem', (req, res) => {
    let x = req.body.x;
    let y = req.body.y;
    console.log(`As coordenadas recebidas foram: (${x}, ${y})`);
    let h = Math.hypot(x, y);
    res.json({ hipotenusa: h });
});

// --- NOVA ROTA: Enviar Imagem ---
// 'foto' é o nome do campo (input) que virá do HTML
app.post('/enviar-imagem', upload.single('foto'), (req, res) => {
    try {
        // Se nenhum arquivo foi enviado
        if (!req.file) {
            return res.status(400).send('Nenhum arquivo foi enviado.');
        }

        console.log(`Imagem recebida com sucesso: ${req.file.filename}`);
        
        // Resposta enviada de volta ao cliente
        res.json({ 
            mensagem: 'Imagem salva com sucesso!',
            nomeArquivo: req.file.filename,
            caminho: req.file.path
        });

    } catch (error) {
        console.error('Erro no upload:', error);
        res.status(500).send('Erro interno ao salvar a imagem.');
    }
});

const obterIP = () => {
    const interfaces = os.networkInterfaces();
    for (let nomeInterface in interfaces) {
        for (let info of interfaces[nomeInterface]) {
            if (info.family === 'IPv4' && !info.internal) return info.address;
        }
    }
    return 'localhost';
};

const ip = obterIP();

app.listen(port, '0.0.0.0', () => {
    console.log(`Servidor rodando em http://${ip}:${port}`);
});
